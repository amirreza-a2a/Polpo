% Constrain oversized images to text block bounds while preserving aspect ratio
\makeatletter
\def\polpoMaxwidth{\ifdim\Gin@nat@width>\linewidth\linewidth\else\Gin@nat@width\fi}
\def\polpoMaxheight{\ifdim\Gin@nat@height>0.75\textheight 0.75\textheight\else\Gin@nat@height\fi}
\makeatother
\setkeys{Gin}{width=\polpoMaxwidth,height=\polpoMaxheight,keepaspectratio}

% Clamps image scaling to available text area, preventing full-height images from creating orphan blank pages
\makeatletter
\ifdefined\pandocbounded
  \ifdefined\pandoc@box\else\newsavebox\pandoc@box\fi
  \renewcommand*\pandocbounded[1]{%
    \sbox\pandoc@box{#1}%
    \Gscale@div\@tempa{\dimexpr\textheight-4\baselineskip\relax}{\dimexpr\ht\pandoc@box+\dp\pandoc@box\relax}%
    \Gscale@div\@tempb{\linewidth}{\wd\pandoc@box}%
    \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi%
    \ifdim\@tempa\p@<\p@\scalebox{\@tempa}{\usebox\pandoc@box}%
    \else\usebox{\pandoc@box}%
    \fi%
  }
\fi
\makeatother

% MathJax compatibility and RTL Persian text inside math mode
\providecommand{\style}[2]{#2}

% XeLaTeX TeX--XeT RTL guards for xdvipdfmx \special nodes
\ifXeTeX
  % 1. Prevent TeX--XeT from reversing \special{x:gsave/scale} and \special{x:grestore} in \includegraphics
  \NewCommandCopy{\polpoOrigIncludegraphics}{\includegraphics}
  \RenewDocumentCommand{\includegraphics}{s O{} m}{%
    \leavevmode\hbox{\beginL
      \IfBooleanTF{#1}{%
        \polpoOrigIncludegraphics*[#2]{#3}%
      }{%
        \polpoOrigIncludegraphics[#2]{#3}%
      }%
    \endL}%
  }

  % 2. Prevent TeX--XeT from reversing PGF \special{pdf:code ...} streams in tcolorbox/callouts and pgfpicture
  \makeatletter
  \NewCommandCopy{\polpoOrigPgfpicture}{\pgfpicture}
  \NewCommandCopy{\polpoOrigEndpgfpicture}{\endpgfpicture}
  \renewcommand{\pgfpicture}{\ifvmode\leavevmode\fi\hbox\bgroup\beginL\polpoOrigPgfpicture}
  \renewcommand{\endpgfpicture}{\polpoOrigEndpgfpicture\endL\egroup}

  % 3. Prevent TeX--XeT from mirroring parentheses in amsmath equation numbers and \tag{...}
  \ifdef{\maketag@@@}{%
    \NewCommandCopy{\polpoOrigMaketag}{\maketag@@@}%
    \renewcommand{\maketag@@@}[1]{\hbox{\beginL\polpoOrigMaketag{#1}\endL}}%
  }{}
  \makeatother
\fi
